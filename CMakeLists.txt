cmake_minimum_required(VERSION 3.20)
project(trading_bot VERSION 0.1.0 LANGUAGES CXX)

option(TRADING_NATIVE "Optimize for this machine (not portable)" OFF)
option(TRADING_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(TRADING_LTO "Enable release link-time optimization" OFF)

add_library(trading_core INTERFACE)
target_include_directories(trading_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(trading_core INTERFACE cxx_std_20)
if(MSVC)
    # Cache-line alignment deliberately pads Tick, the book and queue cursor storage.
    target_compile_options(trading_core INTERFACE /W4 /WX /wd4324 /permissive-)
else()
    target_compile_options(trading_core INTERFACE -Wall -Wextra -Wpedantic -Werror)
    if(TRADING_NATIVE)
        target_compile_options(trading_core INTERFACE -march=native)
    endif()
    if(TRADING_SANITIZERS)
        target_compile_options(trading_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(trading_core INTERFACE -fsanitize=address,undefined)
    endif()
endif()

function(trading_executable name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE trading_core)
    if(TRADING_LTO)
        include(CheckIPOSupported)
        check_ipo_supported(RESULT supported OUTPUT reason)
        if(NOT supported)
            message(FATAL_ERROR "LTO unavailable: ${reason}")
        endif()
        set_property(TARGET ${name} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    endif()
endfunction()

include(CTest)
find_package(Threads REQUIRED)
trading_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE Threads::Threads)
add_test(NAME core COMMAND core_tests)
trading_executable(execution_tests tests/execution_tests.cpp)
add_test(NAME execution COMMAND execution_tests)
trading_executable(hot_path_tests tests/hot_path_tests.cpp)
add_test(NAME hot_path COMMAND hot_path_tests)
trading_executable(paper_trader app/main.cpp)
trading_executable(latency_bench bench/latency.cpp)
add_library(paper_engine SHARED app/paper_api.cpp)
target_link_libraries(paper_engine PRIVATE trading_core)
target_compile_definitions(paper_engine PRIVATE PAPER_ENGINE_BUILD)
trading_executable(api_tests tests/api_tests.cpp)
target_link_libraries(api_tests PRIVATE paper_engine)
add_test(NAME paper_api COMMAND api_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/network-paper.conf)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_replay.py
        $<TARGET_FILE:paper_trader> ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME quote_feed COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_feeds.py)
    # The ordinary Python host does not preload the sanitizer runtime. Native ABI
    # behavior is still checked by the linked api_tests under sanitizers.
    if(NOT TRADING_SANITIZERS)
        add_test(NAME network_paper COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_network_paper.py
            $<TARGET_FILE:paper_engine> ${CMAKE_CURRENT_SOURCE_DIR})
        set_tests_properties(network_paper PROPERTIES TIMEOUT 30)
    endif()
endif()
